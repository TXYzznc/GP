## 1. UI 系统集成

### 1.1 为所有 UIForm 添加音效

在 UIForm 基类中修改：

```csharp
public partial class UIForm : UIFormLogic
{
    private int _openSFXId = AudioClipIds.SFX_UIOpen;
    private int _closeSFXId = AudioClipIds.SFX_UIClose;

    public override void OnOpen(object userData)
    {
        base.OnOpen(userData);
        // 播放打开音效
        if (_openSFXId > 0)
        {
            AudioManager.Instance?.PlaySFX(_openSFXId);
        }
    }

    public override void OnClose()
    {
        // 播放关闭音效
        if (_closeSFXId > 0)
        {
            AudioManager.Instance?.PlaySFX(_closeSFXId);
        }
        base.OnClose();
    }

    // 允许子类覆盖音效 ID
    protected void SetAudioIds(int openId, int closeId)
    {
        _openSFXId = openId;
        _closeSFXId = closeId;
    }
}
```

## 2. 战斗系统集成

### 2.1 战斗管理器集成

```csharp
public class CombatManager
{
    public void StartCombat(CombatConfig config)
    {
        // 根据敌人等级选择 BGM
        int bgmId = config.EnemyLevel switch
        {
            >= 20 => AudioClipIds.BGM_CombatBoss,
            >= 10 => AudioClipIds.BGM_CombatElite,
            _ => AudioClipIds.BGM_CombatNormal
        };

        AudioManager.Instance?.PlayBGM(bgmId);
        AudioManager.Instance?.PlaySFX(AudioClipIds.SFX_CombatStart);
    }

    public void OnCombatEnd(bool isVictory)
    {
        int sfxId = isVictory 
            ? AudioClipIds.SFX_CombatVictory 
            : AudioClipIds.SFX_CombatDefeat;
        
        AudioManager.Instance?.PlaySFX(sfxId);
        AudioManager.Instance?.StopBGM(fadeOutTime: 1.5f);
    }
}
```

### 2.2 棋子攻击音效

```csharp
public class ChessEntity
{
    public void ExecuteAttack(Chess target)
    {
        // 攻击蓄力阶段
        AudioManager.Instance?.PlaySFX(AudioClipIds.SFX_AttackCharge, 0.8f);
        
        // 动画播放中调用（在关键帧处）
        Invoke(nameof(PlayAttackWhoosh), 0.3f);
    }

    private void PlayAttackWhoosh()
    {
        AudioManager.Instance?.PlaySFX(AudioClipIds.SFX_AttackWhoosh);
    }

    public void TakeDamage(int damage, DamageType damageType)
    {
        // 根据伤害大小播放不同的音效
        int sfxId = damage switch
        {
            >= 100 => AudioClipIds.SFX_HitCritical,
            >= 50 => AudioClipIds.SFX_HitHeavy,
            >= 20 => AudioClipIds.SFX_HitNormal,
            _ => AudioClipIds.SFX_HitLight
        };

        AudioManager.Instance?.PlaySFX(sfxId);
    }
}
```

## 3. 探索系统集成

### 3.1 场景切换时的 BGM

```csharp
public class ExploreManager
{
    private Dictionary<int, int> _sceneToBgmMap = new()
    {
        { 10101, AudioClipIds.BGM_Tiangong },
        { 10102, AudioClipIds.BGM_Herheim },
        { 10103, AudioClipIds.BGM_Takamahara },
        // ... 其他副本
    };

    public void EnterScene(int sceneId)
    {
        if (_sceneToBgmMap.TryGetValue(sceneId, out int bgmId))
        {
            AudioManager.Instance?.PlayBGM(bgmId, fadeInTime: 1f);
        }
    }
}
```

```csharp
public class PlayerController

    private float _footstepCooldown = 0.4f;
    private float _lastFootstepTime = 0;

    private void Update()
    {
        if (IsMoving && Time.time - _lastFootstepTime > _footstepCooldown)
        {
            // 2D 模式：直接播放
            AudioManager.Instance?.PlaySFX(AudioClipIds.SFX_Footstep, 0.6f);
            
            // 3D 模式：按位置播放（可选）
            // AudioManager.Instance?.PlaySFX3D(AudioClipIds.SFX_Footstep, transform.position, 0.6f);
            
            _lastFootstepTime = Time.time;
        }
    }
}
```

## 5. 音量设置集成

### 5.1 设置界面

```csharp
public class SettingsUI : UIForm
{
    private Slider _bgmVolumeSlider;
    private Slider _sfxVolumeSlider;
    private Toggle _muteToggle;

    public override void OnOpen(object userData)
    {
        base.OnOpen(userData);
        
        // 初始化滑块值
        _bgmVolumeSlider.value = AudioManager.Instance.GetTrackVolume(AudioTrackType.BGM);
        _sfxVolumeSlider.value = AudioManager.Instance.GetTrackVolume(AudioTrackType.SFX);

        // 监听变化
        _bgmVolumeSlider.onValueChanged.AddListener(OnBGMVolumeChange);
        _sfxVolumeSlider.onValueChanged.AddListener(OnSFXVolumeChange);
        _muteToggle.onValueChanged.AddListener(OnMuteToggle);
    }

    private void OnBGMVolumeChange(float value)
    {
        AudioManager.Instance?.SetTrackVolume(AudioTrackType.BGM, value);
    }

    private void OnSFXVolumeChange(float value)
    {
        AudioManager.Instance?.SetTrackVolume(AudioTrackType.SFX, value);
    }

    private void OnMuteToggle(bool isMuted)
    {
        AudioManager.Instance?.SetMute(isMuted);
    }
}
```

## 常见问题

### Q: 音效没有播放？
**A**: 
1. 检查配置表中的 ResourcePath 是否正确
2. 确认音效文件在正确的路径
3. 查看控制台错误日志

### Q: BGM 切换时有声音断层？
**A**: 
1. 检查淡出淡入时长设置
2. 确保两个 BGM 的音量设置一致

### Q: 如何为特定音效设置 3D 定位？
**A**: 
```csharp
AudioManager.Instance?.PlaySFX3D(
    AudioClipIds.SFX_AttackWhoosh,
    enemyPosition,
    volumeScale: 1f
);
```

### Q: 如何预加载音效避免首次播放延迟？
**A**: 
```csharp
// 在场景加载时预加载
AudioManager.Instance?.PreloadAudioClip(AudioClipIds.SFX_CombatStart);
```

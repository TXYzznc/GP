## I. 需求分析

### 1. 功能需求

#### 1.1 BGM 系统
- [ ] 支持 BGM 切换（带淡入淡出效果）
- [ ] 支持 BGM 暂停/恢复
- [ ] 支持 BGM 循环播放
- [ ] 支持多个 BGM 队列管理（场景过渡时的平滑切换）
- [ ] 支持 BGM 音量独立调节
- [ ] 支持 BGM 播放完成回调

#### 1.2 SFX 系统
- [ ] 支持 SFX 立即播放
- [ ] 支持 SFX 延迟播放
- [ ] 支持 SFX 3D 定位播放（可选）
- [ ] 支持 SFX 音量独立调节
- [ ] 支持 SFX 音量随距离衰减（3D音效）
- [ ] 支持同时播放多个 SFX
- [ ] 支持 SFX 循环播放（环境音）

#### 1.3 音量管理
- [ ] 独立音轨音量控制（BGM、SFX、环境音、语音）
- [ ] 全局音量控制
- [ ] 音量持久化存储
- [ ] 静音功能

#### 1.4 资源管理
- [ ] BGM 流式加载（Streaming）
- [ ] SFX 对象池缓存
- [ ] 音频卸载策略
- [ ] 内存优化

---

### 2. 非功能需求

- **性能**: SFX 播放延迟 < 50ms
- **内存**: 单个 BGM < 5MB，SFX 缓存总量 < 50MB
- **扩展性**: 新增音效不需要改核心代码
- **可配置性**: 所有音效参数可通过 DataTable 配置

### 4.2 BGMManager 框架

```csharp
public class BGMManager : IDisposable
{
    private AudioSource _bgmSource;
    private AudioResourceManager _resourceManager;
    private VolumeManager _volumeManager;
    private Coroutine _fadeCoroutine;
    private Queue<BGMPlayRequest> _bgmQueue = new();

    public BGMManager(AudioResourceManager resourceManager, VolumeManager volumeManager)
    {
        _resourceManager = resourceManager;
        _volumeManager = volumeManager;
        _bgmSource = CreateAudioSource("BGM", isLoop: true, is3D: false);
    }

    public void Play(BGMPlayRequest request)
    {
        if (request.IsImmediate)
        {
            PlayImmediate(request);
        }
        else
        {
            _bgmQueue.Enqueue(request);
            if (!(_fadeCoroutine != null))
            {
                ProcessQueue();
            }
        }
    }

    private void PlayImmediate(BGMPlayRequest request)
    {
        if (_fadeCoroutine != null)
        {
            CoroutineHelper.StopCoroutine(_fadeCoroutine);
        }

        var clip = _resourceManager.GetAudioClip(request.AudioId);
        _bgmSource.clip = clip;
        _bgmSource.volume = _volumeManager.GetEffectiveVolume(AudioTrackType.BGM);
        _bgmSource.Play();

        _fadeCoroutine = CoroutineHelper.StartCoroutine(FadeInCoroutine(request.FadeInTime));
    }

    private void ProcessQueue()
    {
        if (_bgmQueue.Count == 0) return;

        var request = _bgmQueue.Dequeue();
        _fadeCoroutine = CoroutineHelper.StartCoroutine(QueuePlayCoroutine(request));
    }

    private IEnumerator FadeInCoroutine(float duration)
    {
        float elapsed = 0;
        float targetVolume = _volumeManager.GetEffectiveVolume(AudioTrackType.BGM);
        _bgmSource.volume = 0;

        while (elapsed < duration)
        {
            elapsed += Time.deltaTime;
            _bgmSource.volume = Mathf.Lerp(0, targetVolume, elapsed / duration);
            yield return null;
        }

        _bgmSource.volume = targetVolume;
        _fadeCoroutine = null;
    }

    private IEnumerator FadeOutCoroutine(float duration)
    {
        float elapsed = 0;
        float startVolume = _bgmSource.volume;

        while (elapsed < duration)
        {
            elapsed += Time.deltaTime;
            _bgmSource.volume = Mathf.Lerp(startVolume, 0, elapsed / duration);
            yield return null;
        }

        _bgmSource.volume = 0;
        _bgmSource.Stop();
        _fadeCoroutine = null;
    }

    private IEnumerator QueuePlayCoroutine(BGMPlayRequest request)
    {
        // 淡出当前BGM
        if (_bgmSource.isPlaying)
        {
            yield return CoroutineHelper.StartCoroutine(FadeOutCoroutine(request.FadeOutTime));
        }

        // 淡入新BGM
        var clip = _resourceManager.GetAudioClip(request.AudioId);
        _bgmSource.clip = clip;
        _bgmSource.Play();

        yield return CoroutineHelper.StartCoroutine(FadeInCoroutine(request.FadeInTime));

        request.OnComplete?.Invoke();
        ProcessQueue();
    }

    public void Stop(float fadeOutTime = 0.5f)
    {
        if (_fadeCoroutine != null)
        {
            CoroutineHelper.StopCoroutine(_fadeCoroutine);
        }
        _fadeCoroutine = CoroutineHelper.StartCoroutine(FadeOutCoroutine(fadeOutTime));
    }

    public void Pause() => _bgmSource.Pause();
    public void Resume() => _bgmSource.UnPause();

    public void Dispose()
    {
        if (_fadeCoroutine != null)
        {
            CoroutineHelper.StopCoroutine(_fadeCoroutine);
        }
        if (_bgmSource != null)
        {
            Object.Destroy(_bgmSource.gameObject);
        }
    }

    private AudioSource CreateAudioSource(string name, bool isLoop, bool is3D)
    {
        var go = new GameObject($"AudioSource_{name}");
        var source = go.AddComponent<AudioSource>();
        source.loop = isLoop;
        source.spatialBlend = is3D ? 1f : 0f;
        source.priority = 128; // 中等优先级
        return source;
    }
}
```

### 4.3 SFXManager 框架

```csharp
public class SFXManager : IDisposable
{
    private AudioResourceManager _resourceManager;
    private VolumeManager _volumeManager;
    private Dictionary<int, AudioSourcePool> _sfxPools = new();
    private List<Coroutine> _delayedPlayCoroutines = new();

    public SFXManager(AudioResourceManager resourceManager, VolumeManager volumeManager)
    {
        _resourceManager = resourceManager;
        _volumeManager = volumeManager;
    }

    public void Play(SFXPlayRequest request)
    {
        if (request.Delay > 0)
        {
            var coroutine = CoroutineHelper.StartCoroutine(DelayPlayCoroutine(request));
            _delayedPlayCoroutines.Add(coroutine);
            return;
        }

        var pool = GetOrCreatePool(request.AudioId);
        var source = pool.Get();

        if (request.Is3D)
        {
            source.transform.position = request.WorldPosition;
            source.spatialBlend = 1f;
        }
        else
        {
            source.spatialBlend = 0f;
        }

        source.volume = _volumeManager.GetEffectiveVolume(AudioTrackType.SFX) * request.VolumeScale;
        source.PlayOneShot(source.clip);

        // 回收到对象池
        var duration = source.clip.length;
        CoroutineHelper.StartCoroutine(ReturnToPoolCoroutine(pool, source, duration));

        request.OnComplete?.Invoke();
    }

    public void PlayDelayed(int audioId, float delay, float volumeScale = 1f)
    {
        var coroutine = CoroutineHelper.StartCoroutine(
            DelayPlayCoroutine(new SFXPlayRequest { AudioId = audioId, VolumeScale = volumeScale, Delay = delay })
        );
        _delayedPlayCoroutines.Add(coroutine);
    }

    private IEnumerator DelayPlayCoroutine(SFXPlayRequest request)
    {
        yield return new WaitForSeconds(request.Delay);
        request.Delay = 0;
        Play(request);
    }

    private IEnumerator ReturnToPoolCoroutine(AudioSourcePool pool, AudioSource source, float duration)
    {
        yield return new WaitForSeconds(duration);
        pool.Return(source);
    }

    public void Stop(int audioId)
    {
        if (_sfxPools.TryGetValue(audioId, out var pool))
        {
            pool.StopAll();
        }
    }

    public void StopAll()
    {
        foreach (var pool in _sfxPools.Values)
        {
            pool.StopAll();
        }
    }

    private AudioSourcePool GetOrCreatePool(int audioId)
    {
        if (!_sfxPools.TryGetValue(audioId, out var pool))
        {
            var clip = _resourceManager.GetAudioClip(audioId);
            pool = new AudioSourcePool($"SFXPool_{audioId}", clip, initialSize: 5);
            _sfxPools[audioId] = pool;
        }
        return pool;
    }

    public void Dispose()
    {
        foreach (var coroutine in _delayedPlayCoroutines)
        {
            CoroutineHelper.StopCoroutine(coroutine);
        }
        _delayedPlayCoroutines.Clear();

        foreach (var pool in _sfxPools.Values)
        {
            pool.Dispose();
        }
        _sfxPools.Clear();
    }
}
```

### 4.4 VolumeManager 框架

```csharp
public class VolumeManager
{
    private const string VOLUME_PREFS_PREFIX = "Audio_";
    private float _masterVolume = 1f;
    private Dictionary<AudioTrackType, float> _trackVolumes = new()
    {
        { AudioTrackType.BGM, 0.8f },
        { AudioTrackType.SFX, 0.9f },
        { AudioTrackType.Ambient, 0.6f },
        { AudioTrackType.Voice, 1f }
    };
    private bool _isMuted = false;

    public VolumeManager()
    {
        LoadVolumeSettings();
    }

    public void SetMasterVolume(float volume)
    {
        _masterVolume = Mathf.Clamp01(volume);
        PlayerPrefs.SetFloat(VOLUME_PREFS_PREFIX + "Master", _masterVolume);
        ApplyVolume();
    }

    public void SetTrackVolume(AudioTrackType trackType, float volume)
    {
        _trackVolumes[trackType] = Mathf.Clamp01(volume);
        PlayerPrefs.SetFloat(VOLUME_PREFS_PREFIX + trackType, _trackVolumes[trackType]);
        ApplyVolume();
    }

    public float GetTrackVolume(AudioTrackType trackType)
    {
        return _trackVolumes[trackType];
    }

    public float GetEffectiveVolume(AudioTrackType trackType)
    {
        if (_isMuted) return 0;
        return _masterVolume * _trackVolumes[trackType];
    }

    public void SetMute(bool isMuted)
    {
        _isMuted = isMuted;
        PlayerPrefs.SetInt(VOLUME_PREFS_PREFIX + "Muted", isMuted ? 1 : 0);
        ApplyVolume();
    }

    private void LoadVolumeSettings()
    {
        _masterVolume = PlayerPrefs.GetFloat(VOLUME_PREFS_PREFIX + "Master", 1f);
        foreach (var trackType in new[] { AudioTrackType.BGM, AudioTrackType.SFX, AudioTrackType.Ambient, AudioTrackType.Voice })
        {
            _trackVolumes[trackType] = PlayerPrefs.GetFloat(VOLUME_PREFS_PREFIX + trackType, _trackVolumes[trackType]);
        }
        _isMuted = PlayerPrefs.GetInt(VOLUME_PREFS_PREFIX + "Muted", 0) == 1;
    }

    private void ApplyVolume()
    {
        // 所有AudioSource会在下一帧读取 GetEffectiveVolume
    }

    public void Dispose()
    {
        // 保存设置
        PlayerPrefs.Save();
    }
}
```

### 4.5 AudioResourceManager 框架

```csharp
public class AudioResourceManager
{
    private Dictionary<int, DRAudioClipTable> _audioConfigs = new();
    private Dictionary<int, AudioClip> _clipCache = new();
    private Dictionary<string, AudioClip> _streamingClips = new();

    public void InitializeConfigs(DRAudioClipTable[] configs)
    {
        foreach (var config in configs)
        {
            _audioConfigs[config.Id] = config;
        }
    }

    public AudioClip GetAudioClip(int audioId)
    {
        if (_clipCache.TryGetValue(audioId, out var clip))
        {
            return clip;
        }

        if (_audioConfigs.TryGetValue(audioId, out var config))
        {
            clip = Resources.Load<AudioClip>(config.ResourcePath);
            if (config.AudioType == (int)AudioClipType.BGM)
            {
                // BGM使用流式加载，不缓存
                return clip;
            }
            else
            {
                _clipCache[audioId] = clip;
                return clip;
            }
        }

        DebugEx.Error("AudioResourceManager", $"❌ 未找到音效ID: {audioId}");
        return null;
    }

    public void PreloadAudioClip(int audioId)
    {
        if (!_clipCache.ContainsKey(audioId))
        {
            var clip = GetAudioClip(audioId);
            if (clip != null)
            {
                _clipCache[audioId] = clip;
            }
        }
    }

    public void UnloadAudioClip(int audioId)
    {
        if (_clipCache.TryGetValue(audioId, out var clip))
        {
            Resources.UnloadAsset(clip);
            _clipCache.Remove(audioId);
        }
    }

    public void UnloadAllAudioClips()
    {
        var keys = _clipCache.Keys.ToList();
        foreach (var key in keys)
        {
            UnloadAudioClip(key);
        }
        _clipCache.Clear();
    }

    public void Dispose()
    {
        UnloadAllAudioClips();
    }
}
```

### 4.6 AudioSourcePool 对象池

```csharp
public class AudioSourcePool : IDisposable
{
    private string _poolName;
    private AudioClip _clip;
    private Queue<AudioSource> _available = new();
    private HashSet<AudioSource> _inUse = new();

    public AudioSourcePool(string poolName, AudioClip clip, int initialSize = 5)
    {
        _poolName = poolName;
        _clip = clip;

        var parent = new GameObject(_poolName).transform;
        for (int i = 0; i < initialSize; i++)
        {
            var source = CreateAudioSource(parent);
            _available.Enqueue(source);
        }
    }

    public AudioSource Get()
    {
        AudioSource source;
        if (_available.Count > 0)
        {
            source = _available.Dequeue();
        }
        else
        {
            source = CreateAudioSource(null);
        }

        source.clip = _clip;
        source.gameObject.SetActive(true);
        _inUse.Add(source);
        return source;
    }

    public void Return(AudioSource source)
    {
        if (_inUse.Remove(source))
        {
            source.Stop();
            source.gameObject.SetActive(false);
            _available.Enqueue(source);
        }
    }

    public void StopAll()
    {
        foreach (var source in _inUse)
        {
            source.Stop();
        }
    }

    private AudioSource CreateAudioSource(Transform parent)
    {
        var go = new GameObject($"AudioSource_{_poolName}");
        if (parent != null)
            go.transform.SetParent(parent);

        var source = go.AddComponent<AudioSource>();
        source.spatialBlend = 0f; // SFX默认不用3D
        source.priority = 64;
        source.gameObject.SetActive(false);
        return source;
    }

    public void Dispose()
    {
        foreach (var source in _available)
        {
            Object.Destroy(source.gameObject);
        }
        foreach (var source in _inUse)
        {
            Object.Destroy(source.gameObject);
        }
        _available.Clear();
        _inUse.Clear();
    }
}
```

```csharp
public class AudioProcedure : ProcedureBase
{
    public override void OnEnter(ProcedureOwner procedureOwner)
    {
        base.OnEnter(procedureOwner);
        AudioManager.Instance.PlayBGM(10001); // 播放主菜单BGM
    }
```

## VIII. 关键注意事项

### 8.1 时序问题
- ✅ 所有异步加载操作使用 UniTask
- ✅ BGM切换使用协程处理，避免声音突跳
- ✅ SFX 延迟播放要考虑场景卸载的情况

### 8.2 内存优化
- ✅ BGM 使用流式加载（不存储在内存）
- ✅ SFX 使用对象池缓存
- ✅ 定期卸载未使用的 AudioClip

### 8.3 音量管理
- ✅ 每个 AudioSource 设置前都要计算 `GetEffectiveVolume`
- ✅ 音量变化时要逐帧插值（避免突变）
- ✅ 音量设置要持久化存储

### 8.4 错误处理
- ✅ 音效文件不存在时要有降级方案
- ✅ 音效播放异常要记录日志
- ✅ 配置表数据错误要在加载时检查
